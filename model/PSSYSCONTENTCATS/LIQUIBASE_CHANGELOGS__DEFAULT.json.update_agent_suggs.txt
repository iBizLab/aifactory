```

- 返回内容仅包含JSON内容
JSON格式 { "suggs": ["建议问题1", "建议问题2", "建议问题3"], "kbs": [{"id":"ibizaifactory-kb--123", "name":"知识库名称"}] }

<#assign suggs=[]>

<#attempt>
    <#assign ret=utils.json_content(ctx.raw_chat_completion("ibizaifactory-ai--followup", msg)!"{}")?eval>

    <#assign suggs=[]>
    <#assign kbs=[]>

<tool>
<suggestions version="1.0">

   <suggestion type="action" version="1.0">
     <data>
     {"actionid":"open_kb_index@ai_knowledge_base", "action_context": "ai_knowledge_base:${kb_tag!}"}
    </data>
     <metadata>
    {"content_name": "打开${obj.name!}", "action_context": "ai_knowledge_base:${kb_tag!}"}
   </metadata>
   </suggestion>
   </#list>
</#if>
<#if (data[0].id)?? && data[0].id?ends_with("ai_kb_document")>

     <metadata>
    {"content_name": "添加到评论"}
   </metadata>
   </suggestion>

<#list suggs as sugg>

   </metadata>
   </suggestion>
</#list>
 </suggestions>
